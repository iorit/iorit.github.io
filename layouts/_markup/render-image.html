{{- /* 图片渲染 1) 正文里写 {width=400} 时，按 2 倍宽度出图（400 显示宽 / 800 实际像素），保证高分屏不糊； 2) 没写 width 时，超过 1600px 的图仍然缩到 1600px； 3) PNG 会转成 WebP（截屏照片用 PNG 体积是浪费），其它格式保持原样。 想改回纯 PNG：把下面 $toWebp 改成 false。 */ -}} {{- $toWebp := true -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- $w := 0 -}} {{- with index (.Attributes | default dict) "width" -}}{{- $w = int . -}}{{- end -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $r := . -}} {{- if and (eq .MediaType.MainType "image") (ne .MediaType.SubType "svg") (ne .MediaType.SubType "gif") -}} {{- $target := 0 -}} {{- if gt $w 0 -}} {{- $target = mul $w 2 -}} {{- else if gt .Width 1600 -}} {{- $target = 1600 -}} {{- end -}} {{- if and (gt $target 0) (lt $target .Width) -}} {{- $fmt := .MediaType.SubType -}} {{- if and $toWebp (eq .MediaType.SubType "png") -}}{{- $fmt = "webp" -}}{{- end -}} {{- $r = .Resize (printf "%dx %s q82" $target $fmt) -}} {{- end -}} {{- end -}} {{- $src = $r.RelPermalink -}} {{- with $u.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $u.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- end -}} {{- end -}} {{- $attributes := merge .Attributes (dict "alt" .Text "src" $src "title" (.Title | transform.HTMLEscape) "loading" "lazy") -}} {{- /**/ -}}
