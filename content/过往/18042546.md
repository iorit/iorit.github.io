\[\begin{aligned}
H(A\times B)&=\sum_{i=1}^n\sum_{j=1}^n\sum_{k=1}^n(A_{i,k}\times B_{k,j})base^{(i-1)n_+j}\\
&=\sum_{k=1}^n\sum_{i=1}^nA_{i,k}base^{(i-1)n}\sum_{j=1}^nB_{k,j}base^j
